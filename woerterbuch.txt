# ============================================================================
#  OIDA  -  Wörterbuch der Programmiersprache
# ============================================================================
#
#  Links steht das feste Wort, das der Compiler kennt.
#  Rechts schreibst du dein Wort hin:
#
#      IF = WENN
#
#  Regeln (dieselben wie bei OIDA-SQL):
#   - Rechts leer lassen  ->  es gilt das Wort von links.
#   - Mehrere Varianten mit Beistrich:      IF = WENN, WAUN
#   - Rechts dürfen mehrere Wörter stehen:  DECLARE = SOG AU
#     (bei mehreren Wörtern gewinnt immer der längste Treffer)
#   - Groß-/Kleinschreibung ist bei diesen Wörtern egal (wenn = WENN = Wenn).
#   - Zeilen mit # am Anfang sind Kommentare.
#   - Ein Wort darf nur bei EINEM festen Wort stehen
#     (sonst meldet der Compiler einen Fehler und sagt dir, welches).
#   - Die Wörter links darfst du nicht ändern, nur die rechts.
#
#  Was hier als Wort steht, kannst du nicht mehr als Namen für eine Variable
#  oder Funktion verwenden. Zwei Ausnahmen:
#   - Die eingebauten Funktionen (LÄNGN, WURZL ...) gelten nur, wenn gleich
#     eine Klammer folgt. Eine Variable darf also "winkl" heißen.
#   - Direkt hinter einem Punkt ist jedes Wort ein Name: ein Feld oder eine
#     Methode darf "von" oder "ende" heißen (zeitraum.von).
#
#  Welche Datei gilt, zeigt dir:  oida --woerter
#  Diese Datei als UTF-8 speichern (wegen Ö, Ä, Ü, ß).
#
# ============================================================================


# --- Ende einer Anweisung ---------------------------------------------------
OIDA      = OIDA

# --- Variablen --------------------------------------------------------------
DECLARE   = SOG AU, DEKLARIER
# --- Feste Werte, ganz außen:  FIX SCHWERKRAFT = 9.81 OIDA  (gilt überall, auch in Funktionen)
CONST     = FIX

# --- Verzweigung und Schleife -----------------------------------------------
IF        = WENN, WAUN
THEN      = DAUN
ELSE      = SUNST, ANSUNSTN
WHILE     = SOLONG
DO        = TUA, MOCH
END       = PASST SCHO

# --- Zählschleife:  FIA i VON 1 BIS 10 SCHRITT 2 TUA ... PASST SCHO ----------
FOR       = FIA
FROM      = VON
TO        = BIS
STEP      = SCHRITT

# --- Schleife verlassen / gleich zum nächsten Durchlauf ---------------------
BREAK     = HAU OB, VERLOSS
CONTINUE  = WEIDA

# --- Strukturen:  STRUKTUR Punkt  x KOMMAZOI  y KOMMAZOI  PASST SCHO ---------
STRUCT    = STRUKTUR

# --- Listen:  LISTN VON ZOI,  FIA JEDE x IN liste TUA ... PASST SCHO ---------
LIST      = LISTN
EACH      = JEDE, JEDN, JEDS
IN        = IN
ADD       = HAU EINE, DAZUA
REMOVE    = HAU AUSSE

# --- Verzeichnis (Schlüssel -> Wert):  VAZEICHNIS VON KETTN UND ZOI ----------
MAP       = VAZEICHNIS, VERZEICHNIS
# --- Klassen:  KLASSE Gegner ERBT VON Wesen ... PASST SCHO -------------------
#     SÖWA ist das Objekt selbst (SÖWA.lebn), DE OIDN die Elternklasse.
CLASS     = KLASSE
EXTENDS   = ERBT VON
THIS      = SÖWA, MEI
SUPER     = DE OIDN, ELTERN

# --- Fehlende Werte:  SOG AU ziel VIELLEICHT Gegner = NIX OIDA ---------------
#     "name ODA HOID 'neamd'" nimmt den Wert dahinter, wenn vorne NIX steht.
NULL      = NIX
MAYBE     = VIELLEICHT, VILLEICHT, KAUN SEI
ORELSE    = ODA HOID
# --- Fehler:  FUNKTIAUN teil(a ZOI, b ZOI) GIBT ZRUCK ZOI ODA FEHLA -----------
#     PROBIER x = teil(1, 0) ... WONNS SCHIEFGEHT f ... PASST SCHO
ERROR     = FEHLA
TRY       = PROBIER, PROBIERS
CATCH     = WONNS SCHIEFGEHT, WENNS SCHIEFGEHT
# --- Andere Dateien dazunehmen:  BRAUCH "gegner.oida" OIDA,  BRAUCH listen OIDA
IMPORT    = BRAUCH, HOL EINA
# --- Nur in der mitgelieferten Bibliothek: eine Funktion, die in der Laufzeit steckt
EXTERN    = VON DRAUSSN
# --- Funktionen -------------------------------------------------------------
FUNCTION  = FUNKTIAUN
RETURNS   = GIBT ZRUCK
RETURN    = GIB ZRUCK

# --- Ausgabe und Eingabe ----------------------------------------------------
PRINT     = SCHREIB
READ      = LIES

# --- Bedingungen ------------------------------------------------------------
AND       = UND
OR        = ODA
NOT       = NED

# --- Werte ------------------------------------------------------------------
TRUE      = WOAHR, STIMMT
FALSE     = FOISCH, SCHMARRN, STIMMT NED

# --- Datentypen -------------------------------------------------------------
INT       = ZOI
FLOAT     = KOMMAZOI
BOOLEAN   = JONA, WOAHRFOISCH
TEXT      = KETTN, TEXTL

# --- Eingebaute Funktionen für Texte ----------------------------------------
LENGTH    = LÄNGN
SUBSTRING = STICKL
FIND      = SUACH
UPPER     = SCHREI
LOWER     = FLÜSTER
TRIM      = STUTZ
REPLACE   = ERSETZ, AUSTAUSCH
# --- Text in eine Zahl verwandeln; geht das nicht, kommt ein FEHLA zurück ----
PARSEINT   = ZOI AUS
PARSEFLOAT = KOMMAZOI AUS
# --- Eine Zahl als Text mit fester Anzahl von Kommastellen:
#     AUF STÖN(3.14159, 2) ist "3.14"
FIXED      = AUF STÖN, AUF STELLEN
# --- JSON:  OIS JSON(wert) macht aus einem Wert JSON-Text (OIS SCHEES JSON: mit
#     Zeilen und Einrückung), AUS JSON VON Punkt(text) macht aus dem Text
#     wieder einen Wert oder einen FEHLA.
TOJSON       = OIS JSON
TOJSONPRETTY = OIS SCHEES JSON
FROMJSON     = AUS JSON

# --- Eingebaute Funktionen zum Rechnen --------------------------------------
SQRT      = WURZL
POW       = HOCH
ABS       = BETROG
MIN       = KLANSTA
MAX       = GRÖSSTA
SIN       = SINUS
COS       = KOSINUS
ATAN2     = WINKL
EXP       = E_HOCH
FLOOR     = OBRUNDN
CEIL      = AUFRUNDN
ROUND     = RUNDN, PI MOI DAUMEN
RANDOM    = ZUFOI
SEED      = SOMEN

# --- Feste Werte ------------------------------------------------------------
PI        = PI
